(function () {

    const hold = {}; Object.keys(from).forEach((k) => { if (!(k in to)) hold[k] = from[k]; });
    tl.fromTo(target, from, Object.assign({ immediateRender: false }, hold, to), t);
  }
  // seeded random so every render is identical
  let seed = 11; const rnd = () => ((seed = (seed * 16807) % 2147483647) / 2147483647);

  // ------------------------------------------------ stick-figure rig (forward kinematics on nested 0×0 joints)
  // units: torso 110 · upper arm 62 · forearm 58 · thigh 80 · shin 80 · head ⌀72. Angle 0 = limb hanging down,
  // positive = swings toward screen-left (CSS rotate is clockwise). Arms are given relative to the body.
  const U = { T: 110, SH: 100, A: 62, F: 58, L: 80, S: 80, HD: 72 };
  function bone(L, W) { return `<div class="b" style="left:${-W / 2}px;top:${-W / 2}px;width:${W}px;height:${L + W}px;border-radius:${W / 2}px"></div>`; }
  function joint(cls, x, y, inner) { return `<div class="j ${cls}" style="left:${x}px;top:${y}px">${inner}</div>`; }
  function figHTML(W, phone) {
    const hand = phone ? joint('hand', 0, U.F, `<div class="ph" style="top:27px"><div class="phone"></div></div>`) : joint('hand', 0, U.F, '');
    const arm = (s, inner) => joint('a' + s, 0, U.SH, bone(U.A, W) + joint('f' + s, 0, U.A, bone(U.F, W) + inner));
    const leg = (s) => joint('l' + s, 0, 0, bone(U.L, W) + joint('s' + s, 0, U.L, bone(U.S, W)));
    return joint('root', 0, 0,
      leg('L') + leg('R') +
      joint('torso', 0, 0, bone(U.T, W) +
        arm('L', '') + arm('R', hand) +
        joint('neck', 0, U.T, (phone === 'hp' ? `<div style="position:absolute;left:-50px;top:-8px;width:100px;height:100px;border-radius:50%;border:${W}px solid transparent;border-bottom-color:#fff"></div><div style="position:absolute;left:-52px;top:26px;width:20px;height:34px;border-radius:8px;background:var(--accent)"></div><div style="position:absolute;left:32px;top:26px;width:20px;height:34px;border-radius:8px;background:var(--accent)"></div>` : '') + `<div class="head" style="left:${-U.HD / 2}px;top:6px;width:${U.HD}px;height:${U.HD}px;border-width:${W}px"></div>`)));
  }
  const FIGS = {};
  function makeFig(id, parent, o) {   // o: {s, x, feet, flip, W, phone}
    const el = document.createElement('div'); el.className = 'fig' + (id === 'artist' ? ' artist' : ''); el.id = id;
    el.innerHTML = figHTML(o.W || 6, o.phone !== false ? (id === 'artist' ? 'hp' : true) : false); $(parent).appendChild(el);
    const hip = o.feet - (U.L + U.S) * o.s;
    gsap.set(el, { x: o.x, y: hip, scaleX: o.s * (o.flip || 1), scaleY: o.s });
    gsap.set(`#${id} .torso`, { rotation: 180 });
    FIGS[id] = Object.assign({ hip }, o);
    return el;
  }
  const POSE_KEYS = ['root', 'aL', 'fL', 'aR', 'fR', 'lL', 'sL', 'lR', 'sR', 'neck'];
  function rots(P) {
    return { root: P.lean || 0, aL: P.aL[0] - 180, fL: P.aL[1], aR: P.aR[0] - 180, fR: P.aR[1],

  }

  function pose(t, id, P, d, ease) { const r = rots(P); POSE_KEYS.forEach((k) => tl.to(`#${id} .${k}`, { rotation: r[k], duration: d, ease }, t)); }
  // forward kinematics: world point of the right hand's phone centre for figure id in pose P

  const add = (a, b) => [a[0] + b[0], a[1] + b[1]];

  }

  ];
